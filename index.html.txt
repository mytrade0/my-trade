    <!-- ডিরেকশন কন্ট্রোল প্যানেল -->
    <div class="controls">
        <button class="btn btn-up" onclick="setDirection('up')">UP (উপরে যাবে)</button>
        <button class="btn btn-auto" onclick="setDirection('auto')">Auto (স্বাভাবিক)</button>
        <button class="btn btn-down" onclick="setDirection('down')">DOWN (নিচে যাবে)</button>
    </div>

    <script>
        // ১. চার্ট ইনিশিয়ালাইজেশন
        const chartContainer = document.getElementById('chart-container');
        const chart = LightweightCharts.createChart(chartContainer, {
            layout: {
                backgroundColor: '#0f141e',
                textColor: '#d1d4dc',
            },
            grid: {
                vertLines: { color: '#1f293d' },
                horzLines: { color: '#1f293d' },
            },
            timeScale: {
                timeVisible: true,
                secondsVisible: true,
            },
        });

        const candlestickSeries = chart.addCandlestickSeries({
            upColor: '#00e676',
            downColor: '#ff5252',
            borderVisible: false,
            wickUpColor: '#00e676',
            wickDownColor: '#ff5252',
        });

        // রেসপন্সিভ সাইজ
        window.addEventListener('resize', () => {
            chart.applyOptions({ width: chartContainer.clientWidth, height: chartContainer.clientHeight });
        });
        chart.applyOptions({ width: chartContainer.clientWidth, height: chartContainer.clientHeight });

        // ২. প্রারম্ভিক ডেটা ও ভেরিয়েবল
        let lastPrice = 100.00;
        let currentTime = Math.floor(Date.now() / 1000) - 300;
        let currentDirection = 'auto'; // 'up', 'down', or 'auto'

        let initialData = [];
        for (let i = 0; i < 50; i++) {
            let open = lastPrice;
            let close = open + (Math.random() - 0.5) * 2;
            let high = Math.max(open, close) + Math.random();
            let low = Math.min(open, close) - Math.random();
            lastPrice = close;
            currentTime += 5;
            initialData.push({ time: currentTime, open, high, low, close });
        }
        candlestickSeries.setData(initialData);

        // ৩. ডিরেকশন পরিবর্তন ফাংশন
        function setDirection(dir) {
            currentDirection = dir;
        }

        // ৪. রিয়েল-টাইম ক্যান্ডেল আপডেট (প্রতি ১ সেকেন্ডে)
        setInterval(() => {
            let now = Math.floor(Date.now() / 1000);
            let change = (Math.random() * 0.5);

            // ডিরেকশন লজিক
            if (currentDirection === 'up') {
                lastPrice += change;
            } else if (currentDirection === 'down') {
                lastPrice -= change;
            } else {
                lastPrice += (Math.random() - 0.48) * 0.8; // স্বাভাবিক ওঠানামা
            }

            let open = lastPrice - (Math.random() - 0.5) * 0.2;
            let high = Math.max(open, lastPrice) + Math.random() * 0.2;
            let low = Math.min(open, lastPrice) - Math.random() * 0.2;

            document.getElementById('current-price').innerText = lastPrice.toFixed(2);

            candlestickSeries.update({
                time: now,
                open: open,
                high: high,
                low: low,
                close: lastPrice
            });
        }, 1000);
    </script>
